import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { execFileSync } from 'node:child_process';
import { readScholarshipMasterGuide } from '../../work/r07-filtered/scripts/import/ScholarshipMasterGuideReader';
import { scholarshipGuideSection, reviewScholarshipRelationships } from '../../work/r07-filtered/scripts/import/ScholarshipRelationshipSourceReview';

const source = 'workspace/import-sources/scholarships/MANARATAK_Degree_Scholarships_Master_v6_6_2027_Audit_001_010_947.md';
const queue = 'workspace/import-sources/reconciliation/m10-12-14/scholarships-sections.jsonl';
const digest = (value: string | Buffer) => createHash('sha256').update(value).digest('hex');
const bytes = fs.readFileSync(source), artifactHash = digest(bytes);
assert.equal(artifactHash, 'f09c81d8a58c79ff05ef69a6009d181e31fe34d37f2c010ee566cd77e22902fd');
const records = readScholarshipMasterGuide(bytes.toString('utf8'));
const queueBytes = fs.readFileSync(queue);
const decisions = queueBytes.toString('utf8').trim().split(/\r?\n/).map(line => JSON.parse(line));
const byId = new Map(decisions.map(item => [item.sourceRecordId, item]));
assert.equal(records.length, 947); assert.equal(byId.size, 947); assert.equal(decisions.length, 947);
for (const record of records) {
  const decision = byId.get(record.sourceRecordId);
  assert.ok(decision); assert.equal(decision.artifactHash, artifactHash);
  assert.equal(decision.sourceHash, record.sourceHash); assert.equal(decision.sourceLineNumber, record.sourceLineNumber);
  assert.deepEqual(decision.relationshipReview, reviewScholarshipRelationships(record));
  assert.equal(decision.databaseWrites, 0); assert.equal(decision.relationshipReview.importReady, false);
  assert.equal(decision.relationshipReview.canonicalAssignments, 0);
  assert.ok(decision.canonicalQueue.every((item: any) => item.state === 'REVIEW_REQUIRED' && item.canonicalId === null));
}
