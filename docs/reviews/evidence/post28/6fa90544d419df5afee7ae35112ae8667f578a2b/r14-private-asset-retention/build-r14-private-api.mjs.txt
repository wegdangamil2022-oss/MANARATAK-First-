import fs from 'node:fs';import path from 'node:path';import {createRequire} from 'node:module';import {execFileSync} from 'node:child_process';
const work=import.meta.dirname,repo=path.join(work,'r07-filtered'),require=createRequire(path.join(repo,'package.json')),{build}=require('esbuild'),out=path.join(work,'r14-private-api');fs.mkdirSync(out,{recursive:true});
const sourceSha=execFileSync('git',['rev-parse','HEAD'],{cwd:repo,encoding:'utf8'}).trim();
await build({absWorkingDir:repo,entryPoints:['apps/api/src/server.ts'],bundle:true,platform:'node',format:'cjs',external:['uuid','@prisma/client','@node-rs/xxhash','bcrypt','jsonwebtoken','cors','express','helmet','pdfkit','svg-to-pdfkit','fontkit'],sourcemap:true,outfile:path.join(out,'server.cjs'),metafile:true}).then(r=>fs.writeFileSync(path.join(out,'meta.json'),JSON.stringify(r.metafile)));
fs.writeFileSync(path.join(out,'source.json'),JSON.stringify({sourceSha,kind:'SCOPED_LOCAL_DIAGNOSTIC_BUNDLE_NOT_RELEASE_ARTIFACT',apiOnlyBuild:true,lockedDependenciesReused:true,webAdminRebuilt:false})+'\n');console.log('R14_SCOPED_API_BUNDLE_BUILT '+sourceSha);
