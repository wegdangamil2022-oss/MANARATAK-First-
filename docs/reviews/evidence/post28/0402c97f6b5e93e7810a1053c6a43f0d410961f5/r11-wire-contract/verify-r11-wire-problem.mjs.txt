import {legacyErrorToProblem} from './r07-filtered/apps/api/src/presentation/http/ProblemDetails.ts';
import {assertContentionOutcome} from './r07-filtered/scripts/performance/LinuxStudentContention.mjs';
const canonical=legacyErrorToProblem({body:{error:'STUDENT_WORKSPACE_VERSION_CONFLICT'},status:409,instance:'/api/v1/student/workspace',traceId:'R11_SYNTHETIC_CONTRACT'});
