import fs from 'node:fs';import path from 'node:path';import assert from 'node:assert/strict';import crypto from 'node:crypto';import {spawn,spawnSync,execFileSync} from 'node:child_process';
const work=import.meta.dirname,repo=path.join(work,'r07-filtered'),bin=path.join(work,'pg16-binaries/pgsql/bin'),lab=fs.readFileSync(path.join(work,'r14-authenticated-current-lab.txt'),'utf8').trim(),state=JSON.parse(fs.readFileSync(path.join(lab,'lab-state.json'))),hash=b=>crypto.createHash('sha256').update(b).digest('hex');

let started=false,workerCode=1;
try{
assert.equal(spawnSync(path.join(bin,'pg_ctl.exe'),['-D',state.dataDirectory,'-l',path.join(lab,'authenticated-resume-postgres.log'),'-w','start'],{stdio:'ignore',windowsHide:true,timeout:30000}).status,0);started=true;
const actual=execFileSync(path.join(bin,'psql.exe'),['-X','-At','-v','ON_ERROR_STOP=1','-c','SELECT system_identifier::text FROM pg_control_system()'],{env:{...process.env,PGHOST:'127.0.0.1',PGPORT:u.port,PGUSER:u.username,PGPASSWORD:decodeURIComponent(u.password),PGDATABASE:u.pathname.slice(1),PGSSLMODE:'disable'},encoding:'utf8',windowsHide:true}).trim();assert.equal(actual,state.serverIdentity);
const worker=spawn(process.execPath,['--import','tsx',path.join(work,'r14-authenticated-resume-worker.mjs')],{cwd:repo,env:{SystemRoot:process.env.SystemRoot,PATH:path.dirname(process.execPath),NODE_ENV:'test',DATABASE_URL:env.DATABASE_URL,R14_LAB:lab,TSX_TSCONFIG_PATH:path.join(repo,'tsconfig.base.json')},windowsHide:true,stdio:['ignore','pipe','pipe']});let log='';worker.stdout.on('data',b=>{log+=b;process.stdout.write(b);});worker.stderr.on('data',b=>log+=b);workerCode=await new Promise(r=>worker.once('exit',r));fs.writeFileSync(path.join(lab,'authenticated-resume-worker.log'),log.replaceAll(env.DATABASE_URL,'[REDACTED]'));assert.equal(workerCode,0,'RESUMED_CERTIFICATE_WORKER_FAILED');

const file=path.join(lab,'r14-authenticated-resume-result.json'),result=fs.existsSync(file)?JSON.parse(fs.readFileSync(file)):null;fs.writeFileSync(path.join(lab,'authenticated-resume-cleanup.json'),JSON.stringify({ownedIsolatedClusterResumed:true,originalServerIdentityVerified:true,stopped,apiStopped:result?.cleanup.apiStopped===true,providerStopped:result?.cleanup.providerStopped===true,existingNonLabDatabasesTouched:false,workerCode},null,2)+'\n');console.log(JSON.stringify({lab,stopped}));
}
