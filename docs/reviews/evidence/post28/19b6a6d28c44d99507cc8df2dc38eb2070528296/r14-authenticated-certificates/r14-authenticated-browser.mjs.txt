import {execFileSync} from 'node:child_process';import fs from 'node:fs';import path from 'node:path';import crypto from 'node:crypto';import https from 'node:https';import assert from 'node:assert/strict';
import {chromium} from './r07-filtered/node_modules/playwright/index.mjs';
import {totp} from './r07-filtered/packages/infrastructure/src/auth/mfa/Totp.ts';
import {staticArtifactHandler} from './r07-filtered/scripts/release/serve-static-artifact.mjs';
import {gatewayHandler,assertGatewayReadiness} from './r07-filtered/scripts/release/serve-artifact-gateway.mjs';
export async function certificateBrowser({lab,apiOrigin,webOrigin,bundle,owner,admin,mfaSecret,lastStep,certificateId,pdfSha256,redact}){
 const work=import.meta.dirname,builds=JSON.parse(fs.readFileSync(path.join(work,'r14-authenticated-builds.json')));assert.equal(execFileSync('C:/Program Files/Git/cmd/git.exe',['diff','--name-only',builds.sourceSha,bundle.sourceSha,'--','apps/web','apps/admin','packages/application','packages/domain','packages/shared'],{cwd:path.join(work,'r07-filtered'),encoding:'utf8'}).trim(),'');
 for(const unit of ['web','admin']){const dir=builds[unit],index=path.join(dir,'index.html');let html=fs.readFileSync(index,'utf8').replace(/<meta name="manaratak-build-sha" content="[^"]*">/g,'');html=html.replace('</head>','<meta name="manaratak-build-sha" content="'+builds.sourceSha+'"></head>');fs.writeFileSync(index,html);fs.writeFileSync(path.join(dir,'manaratak-build.json'),JSON.stringify({version:1,unit,sourceSha:builds.sourceSha})+'\n');}
 const ca=fs.readFileSync(path.join(lab,'provider.crt')),spki=crypto.createHash('sha256').update(new crypto.X509Certificate(ca).publicKey.export({type:'spki',format:'der'})).digest('base64');
 await assertGatewayReadiness(apiOrigin,bundle.sourceSha);
 const gateway=https.createServer({cert:ca,key:fs.readFileSync(path.join(lab,'provider.key'))},gatewayHandler({apiOrigin,web:staticArtifactHandler({directory:builds.web,unit:'web',sourceSha:builds.sourceSha}),admin:staticArtifactHandler({directory:builds.admin,unit:'admin',sourceSha:builds.sourceSha})}));
 let browser;const network=[],errors=[],downloads=[];
 try{
 await new Promise(r=>gateway.listen(Number(new URL(webOrigin).port),'127.0.0.1',r));
 browser=await chromium.launch({executablePath:'C:/Program Files (x86)/Microsoft/Edge/Application/msedge.exe',headless:true,args:['--disable-background-networking','--ignore-certificate-errors-spki-list='+spki]});
 async function page(){const c=await browser.newContext({ignoreHTTPSErrors:false,acceptDownloads:true});await c.route('**/*',r=>{const u=new URL(r.request().url());return u.origin===webOrigin||['blob:','data:'].includes(u.protocol)?r.continue():r.abort();});const p=await c.newPage();p.on('pageerror',e=>errors.push(redact(e.message)));p.on('response',r=>{const u=new URL(r.url());if(u.origin===webOrigin)network.push({path:u.pathname,status:r.status(),method:r.request().method()});});return p;}
 const student=await page();await student.goto(webOrigin+'/ar/login');await student.locator('#account-email').fill(owner.email);await student.locator('#account-password').fill(owner.password);const login=student.waitForResponse(r=>new URL(r.url()).pathname==='/api/v1/auth/login'&&r.request().method()==='POST');await student.locator('form button[type=submit]').click();assert.equal((await login).status(),200);await student.goto(webOrigin+'/ar/student?tab=learning#certificates');
 const panel=student.locator('#certificates');await panel.waitFor({timeout:20000});
 const previewResponse=student.waitForResponse(r=>new URL(r.url()).pathname==='/api/v1/student/certificates/'+certificateId+'/artifacts/preview/content');await panel.getByRole('button',{name:'عرض الشهادة',exact:true}).click();assert.equal((await previewResponse).status(),200);await panel.locator('img').waitFor();assert(await panel.locator('img').evaluate(img=>img.complete&&img.naturalWidth>0));
 async function download(p,button,who){const waiting=p.waitForEvent('download');await button.click();const d=await waiting;assert.equal(await d.failure(),null);const bytes=fs.readFileSync(await d.path()),sha=crypto.createHash('sha256').update(bytes).digest('hex');assert.equal(sha,pdfSha256);downloads.push({actor:who,sha256:sha,bytes:bytes.length});}
 await download(student,panel.getByRole('button',{name:'تحميل الشهادة PDF',exact:true}),'student');
 await student.screenshot({path:path.join(lab,'authenticated-student-certificate.png'),fullPage:false});
 if(Math.floor(Date.now()/30000)<=lastStep)await new Promise(r=>setTimeout(r,(lastStep+1)*30000-Date.now()+250));
 const operator=await page();await operator.goto(webOrigin+'/admin/login');await operator.locator('#admin-email').fill(admin.email);await operator.locator('#admin-password').fill(admin.password);await operator.locator('#admin-mfa-code').fill(totp(mfaSecret,Math.floor(Date.now()/30000)));const adminLogin=operator.waitForResponse(r=>new URL(r.url()).pathname==='/api/v1/auth/login'&&r.request().method()==='POST');await operator.locator('form button[type=submit]').click();assert.equal((await adminLogin).status(),200);
 await operator.waitForURL(u=>!u.pathname.endsWith('/login'));await operator.goto(webOrigin+'/admin/certificates/'+certificateId);await download(operator,operator.getByRole('button',{name:'تنزيل PDF',exact:true}),'admin');await operator.screenshot({path:path.join(lab,'authenticated-admin-certificate.png'),fullPage:false});
 assert.equal(errors.length,0,errors.join(';'));assert(!network.some(n=>n.status>=500));assert(!network.some(n=>n.path.includes('delivery-grant')));
 const proof={status:'PASS_LOCAL_BUILT_WEB_ADMIN_CERTIFICATE_ACTIONS',apiSourceSha:bundle.sourceSha,webSourceSha:builds.sourceSha,adminSourceSha:builds.sourceSha,unchangedFrontendSourceComparedToApi:true,mixedCommitLocalDiagnostic:true,httpsSpkiPinned:true,ignoreHTTPSErrors:false,externalBrowserRequestsBlocked:true,studentLoginViaUi:true,adminFreshMfaLoginViaUi:true,studentPreviewVisible:true,studentAndAdminPdfDownloadsMatched:true,providerCapabilityEndpointUsed:false,network,pageErrors:errors,downloads};
 fs.writeFileSync(path.join(lab,'authenticated-browser-result.json'),JSON.stringify(proof,null,2)+'\n');return proof;
 }catch(e){fs.writeFileSync(path.join(lab,'authenticated-browser-failure.json'),JSON.stringify({message:redact(e.message),stack:redact(e.stack),network,pageErrors:errors},null,2)+'\n');throw e;}
 finally{await browser?.close();gateway.closeAllConnections();await new Promise(r=>gateway.close(r));}
}
