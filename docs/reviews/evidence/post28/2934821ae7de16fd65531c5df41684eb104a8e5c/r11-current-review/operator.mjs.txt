import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {execFileSync} from 'node:child_process';

import {verifyEvidenceIndex} from './r07-filtered/scripts/lib/verify-evidence-index.mjs';
const work=import.meta.dirname,repo=path.join(work,'r07-filtered');
const sha=execFileSync('git',['rev-parse','HEAD'],{cwd:repo,encoding:'utf8'}).trim();
const dest=path.join(repo,'docs/reviews/evidence/post28',sha,'r11-current-review');

 'docs/reviews/evidence/post28/4b8e22f4cad6c5cdca384cf3e273c6d5cf97d61b/r11-linux/evidence-index.json',
 'docs/reviews/evidence/post28/6d440363adb1477101402f00371e2319c3aae84f/r11-mfa-enrollment/evidence-index.json',

])retained.push({index,...verifyEvidenceIndex(path.join(repo,index),{root:repo,gitRef:'HEAD'})});
